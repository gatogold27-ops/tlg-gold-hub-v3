local function GetParent()
    local p
    pcall(function() if gethui then p = gethui() end end)
    if not p then
        pcall(function()
            if syn and syn.protect_gui then
                local g = Instance.new("ScreenGui")
                syn.protect_gui(g)
                g.Parent = CoreGui
                p = g
            end
        end)
    end
    if not p then
        local ok = pcall(function()
            local f = Instance.new("Folder")
            f.Parent = CoreGui
            f:Destroy()
            return CoreGui
        end)
        if ok then p = CoreGui end
    end
    if not p then p = TLG.LP:WaitForChild("PlayerGui") end
    return p
end

function TLG.PGlow(stroke, minT, maxT, dur)
    minT = minT or 0.2
    maxT = maxT or 0.8
    dur = dur or 1.5
    task.spawn(function()
        while stroke and stroke.Parent do
            local t1 = TweenService:Create(stroke, TweenInfo.new(dur, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Transparency = maxT})
            t1:Play()
            t1.Completed:Wait()
            if not stroke.Parent then break end
            local t2 = TweenService:Create(stroke, TweenInfo.new(dur, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Transparency = minT})
            t2:Play()
            t2.Completed:Wait()
        end
    end)
end

function TLG.GetStats(plr)
    local s = {Level=0, Health=0, MaxHealth=0, Gems=0, Cash=0, Power=0, Melee=0, Weapon=0}
    pcall(function()
        local ch = plr.Character
        if ch then
            local hu = ch:FindFirstChildOfClass("Humanoid")
            if hu then
                s.Health = math.floor(hu.Health)
                s.MaxHealth = math.floor(hu.MaxHealth)
            end
        end
    end)
    local function scan(folder, depth)
        if not folder or depth > 4 then return end
        for _, v in ipairs(folder:GetChildren()) do
            if v:IsA("IntValue") or v:IsA("NumberValue") or v:IsA("StringValue") then
                local n = v.Name:lower()
                local val = tonumber(v.Value) or 0
                if n == "level" or n == "lvl" then s.Level = val
                elseif n == "gem" or n == "gems" then s.Gems = val
                elseif n == "cash" or n == "money" then s.Cash = val
                elseif n == "power" or n == "poder" or n == "memepowerlevel" then s.Power = val
                elseif n == "melee" or n == "meleelevel" then s.Melee = val
                elseif n == "sword" or n == "swordlevel" or n == "weapon" or n == "arma" then s.Weapon = val
                end
            elseif v:IsA("Folder") or v:IsA("Configuration") then
                scan(v, depth + 1)
            end
        end
    end
    for _, n in ipairs({"leaderstats","PlayerData","Stats","Data","Values","SaveData"}) do
        pcall(function() scan(plr:FindFirstChild(n), 0) end)
    end
    return s
end

function TLG.GOSt(p)
    if not p or not p.Parent then return end
    TLG.GOS()
    TLG.GO.T = p; TLG.GO.On = true; TLG.GO.Ang = 0
    task.spawn(function()
        local mc = TLG.LP.Character or TLG.LP.CharacterAdded:Wait()
        local mh = mc:WaitForChild("HumanoidRootPart", 5)
        local mhu = mc:FindFirstChildOfClass("Humanoid")
        if not mh or not mhu then TLG.GOS(); return end
        mh.Anchored = true
        mhu.PlatformStand = true
        TLG.GO.C = RunService.PreSimulation:Connect(function(dt)
            if not TLG.GO.On or not TLG.GO.T or not TLG.GO.T.Parent then TLG.GOS(); return end
            local tc = TLG.GO.T.Character
            if not tc then TLG.GOS(); return end
            local th = tc:FindFirstChild("HumanoidRootPart")
            local thu = tc:FindFirstChildOfClass("Humanoid")
            if not th or not thu or thu.Health <= 0 then TLG.GOS(); return end
            TLG.GO.Ang = TLG.GO.Ang + 5 * dt
            if TLG.GO.Ang > math.pi * 2 then TLG.GO.Ang = TLG.GO.Ang - math.pi * 2 end
            local off = Vector3.new(math.cos(TLG.GO.Ang) * 8, 3, math.sin(TLG.GO.Ang) * 8)
            mh.CFrame = CFrame.new(th.Position + off, th.Position) * CFrame.Angles(0, math.pi, 0)
        end)
    end)
end

end
function TLG.FlyS()
    if TLG.Fly.On then return end
    TLG.Fly.On = true
    task.spawn(function()
        local ch = TLG.LP.Character or TLG.LP.CharacterAdded:Wait()
        local h = ch:WaitForChild("HumanoidRootPart", 5)
        local hu = ch:FindFirstChildOfClass("Humanoid")
        if not h or not hu then TLG.Fly.On = false; return end
        local bv = Instance.new("BodyVelocity", h)
        bv.MaxForce = Vector3.new(1e10, 1e10, 1e10)
        bv.Velocity = Vector3.zero
        local bg = Instance.new("BodyGyro", h)
        bg.MaxTorque = Vector3.new(1e10, 1e10, 1e10)
        bg.P = 9e4
        bg.CFrame = h.CFrame
        hu.PlatformStand = true
        TLG.Fly.C = RunService.RenderStepped:Connect(function()
            if not TLG.Fly.On then TLG.FlyE(); return end
            if not h or not h.Parent then TLG.FlyE(); return end
            if not bv or not bv.Parent then
                bv = Instance.new("BodyVelocity", h)
                bv.MaxForce = Vector3.new(1e10, 1e10, 1e10)
            end
            if not bg or not bg.Parent then
                bg = Instance.new("BodyGyro", h)
                bg.MaxTorque = Vector3.new(1e10, 1e10, 1e10)
                bg.P = 9e4
            end
            if not hu.PlatformStand then hu.PlatformStand = true end
            bg.CFrame = TLG.Cam.CFrame
            local mv = Vector3.zero
            local pm = TLG.LP:FindFirstChild("PlayerScripts")
            if pm then
                local pmod = pm:FindFirstChild("PlayerModule")
                if pmod then
                    local ok, controls = pcall(function()
                        local m = require(pmod)
                        return m:GetControls()
                    end)
                    if ok and controls then
                        local ok2, v = pcall(function() return controls:GetMoveVector() end)
                        if ok2 and v then mv = v end
                    end
                end
            end
            if mv.Magnitude < 0.05 then
                bv.Velocity = Vector3.zero
            else
                local cf = TLG.Cam.CFrame
                local dir = (cf.LookVector * -mv.Z) + (cf.RightVector * mv.X)
                if dir.Magnitude > 0.05 then
                    bv.Velocity = dir.Unit * TLG.Fly.Spd
                else
                    bv.Velocity = Vector3.zero
                end
            end
        end)
    end)
end

do
    Sec(PAb, "🎯 AIMBOT PRO", "🎯")

    TLG.AIM = AIM
    local ANTI = TLG.ANTI_DETECT or {Enabled = true, Jitter = true, RateLimit = true, SmoothTP = true, MaxAttacksPerSec = 20}
    TLG.ANTI_DETECT = ANTI

    local function SaveOrig(plr, part, tipo)
        if not AIM.OriginalData[plr] then AIM.OriginalData[plr] = {} end
        if not AIM.OriginalData[plr][tipo] then
            AIM.OriginalData[plr][tipo] = {Size = part.Size, Transparency = part.Transparency, CanCollide = part.CanCollide, Material = part.Material, Color = part.Color}
        end
    end
    local function ExpandHead(plr)
        if not AIM.ExpandHead then return end
        local ch = plr.Character
        if not ch then return end
        local h = ch:FindFirstChild("Head")
        if not h then return end
        SaveOrig(plr, h, "Head")
        pcall(function()
            h.Size = Vector3.new(AIM.HeadSize, AIM.HeadSize, AIM.HeadSize)
            h.Transparency = 1
            h.CanCollide = false
            h.Material = Enum.Material.Neon
            h.Color = TLG.A()
        end)
    end
    local function ExpandBody(plr)
        if not AIM.ExpandBody then return end
        local ch = plr.Character
        if not ch then return end
        local h = ch:FindFirstChild("HumanoidRootPart")
        if not h then return end
        SaveOrig(plr, h, "Body")
        pcall(function()
            h.Size = Vector3.new(AIM.BodySize, AIM.BodySize, AIM.BodySize)
            h.Transparency = 1
            h.CanCollide = false
        end)
    end
    local function RestoreAll()
        for plr, d in pairs(AIM.OriginalData) do
            if plr.Character then
                for tipo, v in pairs(d) do
                    local p = plr.Character:FindFirstChild(tipo == "Head" and "Head" or "HumanoidRootPart")
                    if p then
                        pcall(function()
                            p.Size = v.Size
                            p.Transparency = v.Transparency
                            p.CanCollide = v.CanCollide
                            p.Material = v.Material
                            p.Color = v.Color
                        end)
                    end
                end
            end
        end
        AIM.OriginalData = {}
    end
    local function FindTarget()
        local b = nil
        local bd = AIM.Range
        local mc = TLG.LP.Character
        if not mc then return nil end
        local mh = mc:FindFirstChild("HumanoidRootPart")
        if not mh then return nil end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= TLG.LP and p.Character then
                local h = p.Character:FindFirstChild("HumanoidRootPart")
                local hu = p.Character:FindFirstChildOfClass("Humanoid")
                if h and hu and hu.Health > 0 then
                    local d = (h.Position - mh.Position).Magnitude
                    if d < bd then bd = d; b = p end
                end
            end
        end
        return b
    end
    local function Atk()
        local mc = TLG.LP.Character
        if not mc then return end
        local t = mc:FindFirstChildOfClass("Tool")
        if t then pcall(function() t:Activate() end) end
    end
    local function TPn(target)
        if not AIM.TeleportNear then return end
        local mc = TLG.LP.Character
        if not mc then return end
        local mh = mc:FindFirstChild("HumanoidRootPart")
        local tc = target.Character
        if not tc then return end
        local th = tc:FindFirstChild("HumanoidRootPart")
        if not mh or not th then return end
        local dist = (th.Position - mh.Position).Magnitude
        if dist <= AIM.TeleDist then return end
        local dir = (th.Position - mh.Position).Unit
        local np = th.Position - dir * AIM.TeleDist
        pcall(function() mh.CFrame = CFrame.new(np, th.Position) end)
    end
    local function FaceTarget(target)
        if not AIM.FaceTarget then return end
        local mc = TLG.LP.Character
        if not mc then return end
        local mh = mc:FindFirstChild("HumanoidRootPart")
        local tc = target.Character
        if not tc then return end
        local th = tc:FindFirstChild("HumanoidRootPart")
        if not mh or not th then return end
        pcall(function()
            mh.CFrame = CFrame.new(mh.Position, Vector3.new(th.Position.X, mh.Position.Y, th.Position.Z))
        end)
    end
    local function Loop()
        if not AIM.On then return end
        local t = FindTarget()
        AIM.Target = t
        if not t then return end
        ExpandHead(t)
        ExpandBody(t)
        FaceTarget(t)
        TPn(t)
        if AIM.AutoAttack then
            local n = os.clock()
            if ANTI.RateLimit then
                if n - AIM.RateTimer >= 1 then AIM.RateCounter = 0; AIM.RateTimer = n end
                if AIM.RateCounter >= ANTI.MaxAttacksPerSec then return end
            end
            local interval = AIM.AttackRate * (ANTI.Jitter and (0.7 + math.random() * 0.6) or 1)
            if n - AIM.LastAttack >= interval then
                AIM.LastAttack = n
                AIM.RateCounter = AIM.RateCounter + 1
                Atk()
            end
        end
    end
    local function Start()
        if AIM.On then return end
        AIM.On = true
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= TLG.LP then ExpandHead(p); ExpandBody(p) end
        end
        AIM.Conn = RunService.Heartbeat:Connect(Loop)
        TLG.PNotif("🎯 Aimbot", "✅ Activado", C.Grn)
    end
    local function Stop()
        if not AIM.On then return end
        AIM.On = false
        if AIM.Conn then pcall(function() AIM.Conn:Disconnect() end); AIM.Conn = nil end
        RestoreAll()
        AIM.Target = nil
        TLG.PNotif("🎯 Aimbot", "❌ Desactivado", C.Rd)
    end
    TLG.AIM_Start = Start
    TLG.AIM_Stop = Stop

    local mb = Instance.new("TextButton", PAb)
    mb.Size = UDim2.new(1,0,0,44)
    mb.BackgroundColor3 = C.BG2
    mb.BackgroundTransparency = 0.05
    mb.Font = Enum.Font.GothamBold
    mb.Text = "🎯 ACTIVAR AIMBOT"
    mb.TextColor3 = C.Yel
    mb.TextSize = 12
    mb.BorderSizePixel = 0
    mb.AutoButtonColor = false
    mb.ZIndex = 5
    Instance.new("UICorner", mb).CornerRadius = UDim.new(0,10)
    local mbS = Instance.new("UIStroke", mb)
    mbS.Thickness = 2
    mbS.Color = C.Yel
    TLG.PGlow(mbS, 0.1, 0.5, 1.5)
    TLG.Ripple(mb)
    mb.MouseButton1Click:Connect(function()
        if AIM.On then
            Stop()
            mb.Text = "🎯 ACTIVAR AIMBOT"
        else
            Start()
            mb.Text = "🛑 DESACTIVAR"
        end
    end)

    Lbl(PAb, "Hitbox 50 studs balanceado", C.Cyan)
    Lbl(PAb, "Auto-attack con jitter", C.Grn)

    Sec(PAb, "⚙️ AJUSTES", "⚙️")
    Sli(PAb, "Head Size", 10, 500, 50, function(v) AIM.HeadSize = v end)
    Sli(PAb, "Body Size", 10, 500, 50, function(v) AIM.BodySize = v end)
    Sli(PAb, "Rango", 500, 10000, 5000, function(v) AIM.Range = v end)
    Sli(PAb, "Distancia TP", 2, 30, 5, function(v) AIM.TeleDist = v end)
    Sli(PAb, "Ataques/seg", 10, 60, 20, function(v) AIM.AttackRate = 1 / v end)
    Tog(PAb, "Auto-Attack", true, function(v) AIM.AutoAttack = v end)
    Tog(PAb, "Teletransporte", true, function(v) AIM.TeleportNear = v end)
    Tog(PAb, "Mirar al enemigo", true, function(v) AIM.FaceTarget = v end)

    Sec(PAb, "🛡 ANTI-DETECT", "🛡")
    Tog(PAb, "Anti-Detección", true, function(v) ANTI.Enabled = v end)
    Tog(PAb, "Jitter", true, function(v) ANTI.Jitter = v end)
    Tog(PAb, "Rate Limit", true, function(v) ANTI.RateLimit = v end)
    Sli(PAb, "Max Ataques/s", 10, 50, 20, function(v) ANTI.MaxAttacksPerSec = v end)
end

local function CF(i, t, cb)
    local b = Instance.new("TextButton", FHo)
    b.Size = UDim2.new(0,55,0,46)
    b.BackgroundColor3 = C.BG2
    b.Text = ""
    b.BorderSizePixel = 0
    b.AutoButtonColor = false
    b.ZIndex = 12
    b.ClipsDescendants = true
    Instance.new("UICorner", b).CornerRadius = UDim.new(0,8)
    local bS = Instance.new("UIStroke", b)
    bS.Thickness = 2
    bS.Color = C.Yel
    bS.Transparency = 0.4
    local iL = Instance.new("TextLabel", b)
    iL.Size = UDim2.new(1,0,0,22)
    iL.Position = UDim2.new(0,0,0,4)
    iL.BackgroundTransparency = 1
    iL.Text = i
    iL.TextSize = 18
    iL.TextColor3 = C.Wh
    iL.Font = Enum.Font.GothamBold
    iL.ZIndex = 13
    local nL = Instance.new("TextLabel", b)
    nL.Size = UDim2.new(1,0,0,10)
    nL.Position = UDim2.new(0,0,1,-14)
    nL.BackgroundTransparency = 1
    nL.Text = t
    nL.TextColor3 = C.Wh
    nL.TextSize = 7
    nL.Font = Enum.Font.GothamBold
    nL.ZIndex = 13
    local function SA(s)
        if s then
            b.BackgroundColor3 = C.Yel
            iL.TextColor3 = C.BG
            nL.TextColor3 = C.BG
            bS.Color = C.Wh
            bS.Transparency = 0
        else
            b.BackgroundColor3 = C.BG2
            iL.TextColor3 = C.Wh
            nL.TextColor3 = C.Wh
            bS.Color = C.Yel
            bS.Transparency = 0.4
        end
    end
    b.MouseButton1Click:Connect(function()
        local ns = cb()
        if ns ~= nil then SA(ns) end
    end)
    SA(false)
    return b
end

            TLG.GOSt(TLG.SelectedTarget)

local fdg, fs, fp = false, nil, nil
FH.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        fdg = true
        fs = i.Position
        fp = FM.Position
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if fdg and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - fs
        FM.Position = UDim2.new(fp.X.Scale, fp.X.Offset + d.X, fp.Y.Scale, fp.Y.Offset + d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        fdg = false
    end
end)
