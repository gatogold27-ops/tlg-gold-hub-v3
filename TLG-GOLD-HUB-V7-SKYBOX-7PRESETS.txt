local function GetParent()
    local p
    pcall(function() if gethui then p = gethui() end end)
    if not p then
        pcall(function()
            if syn and syn.protect_gui then
                local g = Instance.new("ScreenGui")
                syn.protect_gui(g); g.Parent = CoreGui; p = g
            end
        end)
    end
    if not p then
        local ok = pcall(function()
            local f = Instance.new("Folder"); f.Parent = CoreGui; f:Destroy(); return CoreGui
        end)
        if ok then p = CoreGui end
    end
    if not p then p = TLG.LP:WaitForChild("PlayerGui") end
    return p
end

    task.spawn(function()

            local t1 = TweenService:Create(stroke, TweenInfo.new(dur, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Transparency = maxT})

local function DeepScan(inst, s, depth)
    if not inst or depth > 6 then return end
    for _, v in ipairs(inst:GetChildren()) do
        local cn = v.ClassName
        if cn == "IntValue" or cn == "NumberValue" or cn == "StringValue" then
            local n = v.Name:lower():gsub("[%s_%-%.]", "")
            local num = tonumber(v.Value)
            if num then
                if     n == "level" or n == "lvl"                                       then s.Level   = num
                elseif n == "gem" or n == "gems"                                        then s.Gems    = num
                elseif n == "cash" or n == "money" or n == "coins" or n == "belly"      then s.Cash    = num
                elseif n == "power" or n == "memepower" or n == "memepowerlevel" or n == "poder" then s.Power = num
                elseif n == "melee" or n == "meleelevel" or n == "fist" or n == "fistslevel" then s.Melee = num
                elseif n == "sword" or n == "swordlevel" or n == "weapon" or n == "weaponlevel" or n == "gun" then s.Weapon = num
                elseif n == "rebirth" or n == "rebirths" or n == "prestige"             then s.Rebirths = num
                elseif n == "rank" or n == "ranklevel"                                  then s.Rank    = num
                elseif n == "kills" or n == "killcount"                                 then s.Kills   = num
                elseif n == "deaths" or n == "deathcount"                               then s.Deaths  = num
                elseif n == "essence" or n == "shards"                                  then s.Essence = num
                elseif n == "tokens" or n == "tickets"                                  then s.Tokens  = num
                end
            end
        elseif cn == "Folder" or cn == "Configuration" or cn == "Model" then
            DeepScan(v, s, depth + 1)
        end
    end
end

function TLG.GetStats(plr)
    local s = {
        Level=0, Health=0, MaxHealth=0, Gems=0, Cash=0, Power=0,
        Melee=0, Weapon=0, Rebirths=0, Rank=0, Kills=0, Deaths=0,
        Essence=0, Tokens=0
    }
    -- Vida (Humanoid, lo más fiable)
    pcall(function()
        local ch = plr.Character
        if ch then
            local hu = ch:FindFirstChildOfClass("Humanoid")
            if hu then
                s.Health = math.floor(hu.Health)
                s.MaxHealth = math.floor(hu.MaxHealth)
            end
        end
    end)
    -- ★ CLAVE MEME SEA: PlayerData recursivo
    pcall(function()
        local pd = plr:FindFirstChild("PlayerData")
        if pd then DeepScan(pd, s, 0) end
    end)
    -- Fallbacks
    for _, n in ipairs({"leaderstats","Stats","Data","Values","SaveData","PlayerStats"}) do
        pcall(function() DeepScan(plr:FindFirstChild(n), s, 0) end)
    end
    -- Attributes del personaje (por si Meme Sea los usa)
    pcall(function()
        local ch = plr.Character
        if ch then
            for _, attr in ipairs({"Level","Power","Melee","Weapon","Gems","Cash","SwordLevel"}) do
                local a = ch:GetAttribute(attr)
                if type(a) == "number" then
                    if attr == "SwordLevel" then s.Weapon = a
                    else s[attr] = a end
                end
            end
        end
    end)
    return s
end

local function AIM_SaveOrig(plr, part, tipo)
    if not AIM.OriginalData[plr] then AIM.OriginalData[plr] = {} end
    if not AIM.OriginalData[plr][tipo] then
        AIM.OriginalData[plr][tipo] = {
            Size = part.Size, Transparency = part.Transparency,
            CanCollide = part.CanCollide, Material = part.Material, Color = part.Color
        }
    end
end
local function AIM_ExpandHead(plr)
    if not AIM.ExpandHead then return end
    local ch = plr.Character; if not ch then return end
    local h = ch:FindFirstChild("Head"); if not h then return end
    AIM_SaveOrig(plr, h, "Head")
    pcall(function()
        h.Size = Vector3.new(AIM.HeadSize, AIM.HeadSize, AIM.HeadSize)
        h.Transparency = 1; h.CanCollide = false
        h.Material = Enum.Material.Neon; h.Color = TLG.A()
    end)
end
local function AIM_ExpandBody(plr)
    if not AIM.ExpandBody then return end
    local ch = plr.Character; if not ch then return end
    local h = ch:FindFirstChild("HumanoidRootPart"); if not h then return end
    AIM_SaveOrig(plr, h, "Body")
    pcall(function()
        h.Size = Vector3.new(AIM.BodySize, AIM.BodySize, AIM.BodySize)
        h.Transparency = 1; h.CanCollide = false
    end)
end
local function AIM_RestoreAll()
    for plr, d in pairs(AIM.OriginalData) do
        if plr.Character then
            for tipo, v in pairs(d) do
                local p = plr.Character:FindFirstChild(tipo == "Head" and "Head" or "HumanoidRootPart")
                if p then
                    pcall(function()
                        p.Size = v.Size; p.Transparency = v.Transparency
                        p.CanCollide = v.CanCollide; p.Material = v.Material; p.Color = v.Color
                    end)
                end
            end
        end
    end
    AIM.OriginalData = {}
end
local function AIM_HasLOS(fromPos, targetPart)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {TLG.LP.Character, targetPart.Parent}
    params.FilterType = Enum.RaycastFilterType.Exclude
    local result = workspace:Raycast(fromPos, targetPart.Position - fromPos, params)
    return result == nil
end
local function AIM_FindTarget()
    local b = nil
    local bd = AIM.Range
    local mc = TLG.LP.Character; if not mc then return nil end
    local mh = mc:FindFirstChild("HumanoidRootPart"); if not mh then return nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= TLG.LP and p.Character then
            if not (AIM.TeamCheck and TLG.LP.Team and p.Team == TLG.LP.Team) then
                local h = p.Character:FindFirstChild("HumanoidRootPart")
                local hu = p.Character:FindFirstChildOfClass("Humanoid")
                if h and hu and hu.Health > 0 then
                    local d = (h.Position - mh.Position).Magnitude
                    if d < bd then
                        if not AIM.WallCheck or AIM_HasLOS(mh.Position, h) then
                            bd = d; b = p
                        end
                    end
                end
            end
        end
    end
    return b
end
local function AIM_Atk()
    local mc = TLG.LP.Character; if not mc then return end
    local t = mc:FindFirstChildOfClass("Tool")
    if t then pcall(function() t:Activate() end) end
end
local function AIM_TPn(target)
    if not AIM.TeleportNear then return end
    local mc = TLG.LP.Character; if not mc then return end
    local mh = mc:FindFirstChild("HumanoidRootPart")
    local tc = target.Character; if not tc then return end
    local th = tc:FindFirstChild("HumanoidRootPart")
    if not mh or not th then return end
    local dist = (th.Position - mh.Position).Magnitude
    if dist <= AIM.TeleDist then return end
    local dir = (th.Position - mh.Position).Unit
    local np = th.Position - dir * AIM.TeleDist
    pcall(function() mh.CFrame = CFrame.new(np, th.Position) end)
end
local function AIM_FaceTarget(target)
    if not AIM.FaceTarget then return end
    local mc = TLG.LP.Character; if not mc then return end
    local mh = mc:FindFirstChild("HumanoidRootPart")
    local tc = target.Character; if not tc then return end
    local th = tc:FindFirstChild("HumanoidRootPart")
    if not mh or not th then return end
    local aimPos = th.Position
    if AIM.Prediction then
        local vel = th.AssemblyLinearVelocity or Vector3.zero
        local ping = TLG.GetPing() / 1000
        aimPos = aimPos + vel * (ping * 0.5)
    end
    pcall(function()
        mh.CFrame = CFrame.new(mh.Position, Vector3.new(aimPos.X, mh.Position.Y, aimPos.Z))
    end)
end
local function AIM_Loop()
    if not AIM.On then return end
    local t = AIM_FindTarget()
    AIM.Target = t
    if not t then return end
    AIM_ExpandHead(t); AIM_ExpandBody(t)
    AIM_FaceTarget(t); AIM_TPn(t)
    if AIM.AutoAttack then
        local n = os.clock()
        if ANTI.RateLimit then
            if n - AIM.RateTimer >= 1 then AIM.RateCounter = 0; AIM.RateTimer = n end
            if AIM.RateCounter >= ANTI.MaxAttacksPerSec then return end
        end
        local interval = AIM.AttackRate * (ANTI.Jitter and (0.7 + math.random() * 0.6) or 1)
        if n - AIM.LastAttack >= interval then
            AIM.LastAttack = n
            AIM.RateCounter = AIM.RateCounter + 1
            AIM_Atk()
        end
    end
end
function TLG.AIM_Start()
    if AIM.On then return end
    AIM.On = true
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= TLG.LP then AIM_ExpandHead(p); AIM_ExpandBody(p) end
    end
    AIM.Conn = RunService.Heartbeat:Connect(AIM_Loop)
    TLG.PNotif("🎯 Aimbot", "✅ Activado", C.Grn)
end
function TLG.AIM_Stop()
    if not AIM.On then return end
    AIM.On = false
    if AIM.Conn then pcall(function() AIM.Conn:Disconnect() end); AIM.Conn = nil end
    AIM_RestoreAll()
    AIM.Target = nil
    TLG.PNotif("🎯 Aimbot", "❌ Desactivado", C.Rd)
end

function TLG.GOSt(p)

    task.spawn(function()
        local mc = TLG.LP.Character or TLG.LP.CharacterAdded:Wait()

        if not mh or not mhu then TLG.GOS(); return end
        mh.Anchored = true; mhu.PlatformStand = true
        TLG.GO.C = RunService.PreSimulation:Connect(function(dt)
            if not TLG.GO.On or not TLG.GO.T or not TLG.GO.T.Parent then TLG.GOS(); return end

            TLG.GO.Ang = TLG.GO.Ang + 5 * dt
            if TLG.GO.Ang > math.pi * 2 then TLG.GO.Ang = TLG.GO.Ang - math.pi * 2 end
            local off = Vector3.new(math.cos(TLG.GO.Ang) * 8, 3, math.sin(TLG.GO.Ang) * 8)
            mh.CFrame = CFrame.new(th.Position + off, th.Position) * CFrame.Angles(0, math.pi, 0)

TLG.Fly = {On = false, Spd = 15000, C = nil, Noclip = true, Keys = {Up = false, Down = false}}
local function FlyCleanupPart(h)
    if not h then return end
    pcall(function()
        for _, n in ipairs({"TLG_FlyBV", "TLG_FlyBG", "BodyVelocity", "BodyGyro"}) do
            local o = h:FindFirstChild(n); if o then o:Destroy() end
        end
    end)
end
function TLG.FlyE()
    TLG.Fly.On = false
    TLG.Fly.Keys.Up = false; TLG.Fly.Keys.Down = false
    local ch = TLG.LP.Character
    if ch then
        local h = ch:FindFirstChild("HumanoidRootPart")
        local hu = ch:FindFirstChildOfClass("Humanoid")
        FlyCleanupPart(h)
        if hu then pcall(function() hu.PlatformStand = false end) end
        if TLG.Fly.Noclip then
            for _, p in ipairs(ch:GetDescendants()) do
                if p:IsA("BasePart") then pcall(function() p.CanCollide = true end) end
            end
        end
    end
    if TLG.Fly.C then pcall(function() TLG.Fly.C:Disconnect() end); TLG.Fly.C = nil end
end
local function FlyGetMoveVector()
    local mv = Vector3.zero
    local pm = TLG.LP:FindFirstChild("PlayerScripts")
    if pm then
        local pmod = pm:FindFirstChild("PlayerModule")
        if pmod then
            local ok, controls = pcall(function()
                local m = require(pmod); return m:GetControls()
            end)
            if ok and controls then
                local ok2, v = pcall(function() return controls:GetMoveVector() end)
                if ok2 and v then mv = v end
            end
        end
    end
    return mv
end
function TLG.FlyS()
    if TLG.Fly.On then return end
    TLG.Fly.On = true
    task.spawn(function()
        local ch = TLG.LP.Character or TLG.LP.CharacterAdded:Wait()
        local h = ch:WaitForChild("HumanoidRootPart", 5)
        local hu = ch:FindFirstChildOfClass("Humanoid")
        if not h or not hu then TLG.Fly.On = false; return end
        FlyCleanupPart(h)
        local bv = Instance.new("BodyVelocity")
        bv.Name = "TLG_FlyBV"; bv.MaxForce = Vector3.new(1e10, 1e10, 1e10)
        bv.Velocity = Vector3.zero; bv.Parent = h
        local bg = Instance.new("BodyGyro")
        bg.Name = "TLG_FlyBG"; bg.MaxTorque = Vector3.new(1e10, 1e10, 1e10)
        bg.P = 9e4; bg.CFrame = h.CFrame; bg.Parent = h
        hu.PlatformStand = true
        TLG.Fly.C = RunService.RenderStepped:Connect(function()
            if not TLG.Fly.On then TLG.FlyE(); return end
            local curCh = TLG.LP.Character; if not curCh then return end
            h = curCh:FindFirstChild("HumanoidRootPart")
            hu = curCh:FindFirstChildOfClass("Humanoid")
            if not h or not hu then return end
            bv = h:FindFirstChild("TLG_FlyBV")
            bg = h:FindFirstChild("TLG_FlyBG")
            if not bv then
                bv = Instance.new("BodyVelocity"); bv.Name = "TLG_FlyBV"
                bv.MaxForce = Vector3.new(1e10, 1e10, 1e10); bv.Parent = h
            end
            if not bg then
                bg = Instance.new("BodyGyro"); bg.Name = "TLG_FlyBG"
                bg.MaxTorque = Vector3.new(1e10, 1e10, 1e10); bg.P = 9e4; bg.Parent = h
            end
            if not hu.PlatformStand then hu.PlatformStand = true end
            if TLG.Fly.Noclip then
                for _, p in ipairs(curCh:GetDescendants()) do
                    if p:IsA("BasePart") then p.CanCollide = false end
                end
            end
            local cam = TLG.Cam or workspace.CurrentCamera
            if cam then bg.CFrame = cam.CFrame end
            local mv = FlyGetMoveVector()
            local dir = Vector3.zero
            if cam then dir = (cam.CFrame.LookVector * -mv.Z) + (cam.CFrame.RightVector * mv.X) end
            local y = 0
            if TLG.Fly.Keys.Up then y = y + 1 end
            if TLG.Fly.Keys.Down then y = y - 1 end
            if y ~= 0 then dir = dir + Vector3.new(0, y, 0) end
            bv.Velocity = dir.Magnitude > 0.05 and dir.Unit * TLG.Fly.Spd or Vector3.zero
        end)
    end)
end

local ConfigFile = "TLG_Config.json"
function TLG.SaveConfig()
    local data = {
        Theme = TLG.CurrentTheme,
        Fly = {Spd = TLG.Fly.Spd, Noclip = TLG.Fly.Noclip},
        Speed = {Val = TLG.Speed.Val},
        Aim = {
            HeadSize = AIM.HeadSize, BodySize = AIM.BodySize,
            Range = AIM.Range, AutoAttack = AIM.AutoAttack,
            TeleportNear = AIM.TeleportNear, TeleDist = AIM.TeleDist,
            FaceTarget = AIM.FaceTarget, TeamCheck = AIM.TeamCheck,
            WallCheck = AIM.WallCheck, Prediction = AIM.Prediction,
            FOVCircle = AIM.FOVCircle
        },
        AutoPrompt = {On = AutoPrompt.On, Range = AutoPrompt.Range},
    }
    local ok = pcall(function()
        if writefile then writefile(ConfigFile, HttpService:JSONEncode(data)) end
    end)
    if ok then TLG.PNotif("💾 Config", "Guardado correctamente", C.Grn)
    else TLG.PNotif("💾 Config", "No se pudo guardar", C.Rd) end
end
function TLG.LoadConfig()
    if not (isfile and isfile(ConfigFile)) then return end
    local ok, data = pcall(function() return HttpService:JSONDecode(readfile(ConfigFile)) end)
    if not ok or not data then return end
    if data.Theme and Themes[data.Theme] then TLG.CurrentTheme = data.Theme end
    if data.Fly then
        TLG.Fly.Spd = data.Fly.Spd or TLG.Fly.Spd
        TLG.Fly.Noclip = data.Fly.Noclip ~= false
    end
    if data.Speed then TLG.Speed.Val = data.Speed.Val or TLG.Speed.Val end
    if data.Aim then for k, v in pairs(data.Aim) do AIM[k] = v end end
    if data.AutoPrompt then AutoPrompt.Range = data.AutoPrompt.Range or AutoPrompt.Range end
    TLG.PNotif("💾 Config", "Cargado", C.Cyan)
end

    kn.Position = UDim2.new((sv-mn)/(mx-mn),-6,0.5,-6)
    kn.BackgroundColor3 = C.Wh; kn.BorderSizePixel = 0; kn.ZIndex = 8

    tn.BackgroundTransparency = 1; tn.Text = ""; tn.ZIndex = 10
    local dragging = false
    local function up(i)
        local pp = math.clamp((i.Position.X - b.AbsolutePosition.X) / b.AbsoluteSize.X, 0, 1)
        local v = math.floor(mn + (mx - mn) * pp)
        fi.Size = UDim2.new(pp,0,1,0)
        kn.Position = UDim2.new(pp,-6,0.5,-6)
        vl.Text = tostring(v)
        pcall(cb, v)
    end
    tn.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; up(i)
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            up(i)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

        {name = "MrBeast Island", aliases = {"MrBeast Island", "MrBeastIsland", "MrBeast"}},
        {name = "Noob Arena", aliases = {"Noob Arena", "NoobArena", "Noob"}},
        {name = "PVP Arena", aliases = {"PVP Arena", "PVPArena", "PvP Arena", "PVP"}},

    local function FindPartFrom(inst)
        if not inst then return nil end
        if inst:IsA("BasePart") then return inst end
        if inst:IsA("Model") then
            if inst.PrimaryPart then return inst.PrimaryPart end
            local p = inst:FindFirstChildWhichIsA("BasePart", true)
            if p then return p end
        end
        return inst:FindFirstChildWhichIsA("BasePart", true)
    end
    local function FindIsland(aliases)
        for _, alias in ipairs(aliases) do
            local fd = workspace:FindFirstChild(alias)
            if fd then
                local part = FindPartFrom(fd)
                if part then return part, alias end
            end
        end
        for _, alias in ipairs(aliases) do
            local lower = alias:lower()
            for _, d in ipairs(workspace:GetDescendants()) do
                if (d:IsA("Model") or d:IsA("Folder") or d:IsA("BasePart")) and d.Name:lower() == lower then
                    local part = FindPartFrom(d)
                    if part then return part, d.Name end
                end
            end
        end
        return nil, nil
    end
    local function TPTo(entry)
        local ch = TLG.LP.Character
        if not ch then TLG.PNotif("🏝️", "Sin personaje", C.Rd); return end
        local h = ch:FindFirstChild("HumanoidRootPart")
        if not h then TLG.PNotif("🏝️", "Sin HRP", C.Rd); return end
        local part, foundName = FindIsland(entry.aliases)
        if part then
            h.CFrame = CFrame.new(part.Position + Vector3.new(0, 60, 0))
            TLG.PNotif("🏝️", "TP → " .. (foundName or entry.name), C.Cyan)
        else
            TLG.PNotif("🏝️", "No encontrada: " .. entry.name, C.Rd)
        end
    end

    end

        end

local NoFall = {On = false, Conn = nil}
function TLG.NoFallStart()
    if NoFall.On then return end
    NoFall.On = true
    local function hook(hu)
        if not hu then return end
        if NoFall.Conn then pcall(function() NoFall.Conn:Disconnect() end) end
        NoFall.Conn = hu.StateChanged:Connect(function(_, new)
            if not NoFall.On then return end
            if new == Enum.HumanoidStateType.Freefall or new == Enum.HumanoidStateType.FallingDown then
                pcall(function() hu:ChangeState(Enum.HumanoidStateType.GettingUp) end)
            end
        end)
    end
    local ch = TLG.LP.Character
    if ch then hook(ch:FindFirstChildOfClass("Humanoid")) end
    TLG.LP.CharacterAdded:Connect(function(c)
        task.wait(0.3)
        if NoFall.On then hook(c:FindFirstChildOfClass("Humanoid")) end
    end)
end
function TLG.NoFallStop()
    NoFall.On = false
    if NoFall.Conn then pcall(function() NoFall.Conn:Disconnect() end); NoFall.Conn = nil end
end

local OrbitalMap = {On = false, Distance = 250, Height = 80, Speed = 35, Angle = 0, Connection = nil}
local function OrbitalRoot()
    local ch = TLG.LP and TLG.LP.Character
    return ch and ch:FindFirstChild("HumanoidRootPart")
end
function TLG.OrbitalMapStart()
    if OrbitalMap.On then return end
    local root = OrbitalRoot(); if not root then return end
    OrbitalMap.On = true; OrbitalMap.Angle = 0
    local center = root.Position
    OrbitalMap.Connection = RunService.RenderStepped:Connect(function(dt)
        if not OrbitalMap.On then return end
        local cur = OrbitalRoot()
        if not cur or not cur.Parent then TLG.OrbitalMapStop(); return end
        OrbitalMap.Angle = OrbitalMap.Angle + math.rad(OrbitalMap.Speed) * dt
        local x = math.cos(OrbitalMap.Angle) * OrbitalMap.Distance
        local z = math.sin(OrbitalMap.Angle) * OrbitalMap.Distance
        local target = center + Vector3.new(x, OrbitalMap.Height, z)
        cur.CFrame = CFrame.lookAt(target, center)
    end)
end
function TLG.OrbitalMapStop()
    OrbitalMap.On = false
    if OrbitalMap.Connection then pcall(function() OrbitalMap.Connection:Disconnect() end); OrbitalMap.Connection = nil end
end
function TLG.OrbitalMapSetDistance(v) OrbitalMap.Distance = math.clamp(tonumber(v) or 250, 25, 2000) end

function TLG.OrbitalMapSetSpeed(v)    OrbitalMap.Speed    = math.clamp(tonumber(v) or 35, 1, 20000) end

local TracerConn = nil
local function UpdateTracers()

        if TracerConn then TracerConn:Disconnect(); TracerConn = nil end

    end

    TracerConn = RunService.RenderStepped:Connect(function()

local function ESPDestroy(plr)
    local pack = ESP.Objects[plr]

    ESP.Objects[plr] = nil
end

local function ESPUpdate(plr)
    if not ESP.On then return end
    local pack = ESP.Objects[plr]
    local ch = plr.Character
    local root = ch and ch:FindFirstChild("HumanoidRootPart")
    local head = ch and ch:FindFirstChild("Head")
    local hum = ch and ch:FindFirstChildOfClass("Humanoid")

    if not ch or not root or not head or not hum then
        ESPCreate(plr); return
    end
    if not pack or not pack.Highlight or not pack.Highlight.Parent then
        ESPCreate(plr); pack = ESP.Objects[plr]
    end
    if not pack then return end

    local color = TLG.A()
    pack.Highlight.FillColor = color
    pack.Highlight.OutlineColor = color
    pack.Highlight.FillTransparency = ESP.Fill and ESP.FillTransparency or 1
    pack.Label.TextColor3 = color
    pack.BoxStroke.Color = color
    pack.Tracer.BackgroundColor3 = color

    local mh = TLG.LP.Character and TLG.LP.Character:FindFirstChild("HumanoidRootPart")
    local distance = mh and math.floor((mh.Position - root.Position).Magnitude) or 0

    if distance > ESP.MaxDistance then
        pack.Billboard.Enabled = false
        pack.Highlight.Enabled = false
        pack.Box.Visible = false
        pack.Tracer.Visible = false
        return
    end
    pack.Billboard.Enabled = true
    pack.Highlight.Enabled = true
    pack.Billboard.MaxDistance = ESP.MaxDistance

    local lines = {}
    if ESP.Names then table.insert(lines, plr.DisplayName .. "  @" .. plr.Name) end
    if ESP.Distance then table.insert(lines, "📏 " .. distance .. " studs") end
    if ESP.Health then
        local hp = math.floor(hum.Health); local maxhp = math.max(1, math.floor(hum.MaxHealth))
        table.insert(lines, "❤ " .. hp .. "/" .. maxhp)
    end
    pack.Label.Text = table.concat(lines, "\n")

    if ESP.Box then
        local cam = workspace.CurrentCamera
        if cam then
            local top = head.Position + Vector3.new(0, head.Size.Y / 2 + 0.5, 0)
            local bottom = root.Position - Vector3.new(0, root.Size.Y / 2, 0)
            local topPos, topOn = cam:WorldToViewportPoint(top)
            local botPos, botOn = cam:WorldToViewportPoint(bottom)
            if topOn and botOn then
                local height = math.abs(botPos.Y - topPos.Y)
                local width = height * 0.55
                pack.Box.Size = UDim2.new(0, width, 0, height)
                pack.Box.Position = UDim2.new(0, topPos.X - width/2, 0, topPos.Y)
                pack.Box.Visible = true
            else
                pack.Box.Visible = false
            end
        end
    else
        pack.Box.Visible = false
    end

    pack.Tracer.Visible = ESP.Tracer
end

function TLG.ESPStart()
    if ESP.On then return end
    ESP.On = true
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= TLG.LP then pcall(ESPCreate, plr) end
    end
    ESP.PlayerAdded = Players.PlayerAdded:Connect(function(plr)
        if ESP.On then task.wait(0.15); pcall(ESPCreate, plr) end
    end)
    ESP.PlayerRemoving = Players.PlayerRemoving:Connect(function(plr) ESPDestroy(plr) end)
    ESP.CharacterConnections = ESP.CharacterConnections or {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= TLG.LP then
            ESP.CharacterConnections[plr] = plr.CharacterAdded:Connect(function()
                if ESP.On then task.wait(0.25); pcall(ESPCreate, plr) end
            end)
        end
    end
    ESP.Render = RunService.RenderStepped:Connect(function()
        if not ESP.On then return end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= TLG.LP then pcall(ESPUpdate, plr) end
        end
    end)
    UpdateTracers()
end

function TLG.ESPStop()
    ESP.On = false
    if ESP.PlayerAdded then pcall(function() ESP.PlayerAdded:Disconnect() end); ESP.PlayerAdded = nil end
    if ESP.PlayerRemoving then pcall(function() ESP.PlayerRemoving:Disconnect() end); ESP.PlayerRemoving = nil end
    if ESP.Render then pcall(function() ESP.Render:Disconnect() end); ESP.Render = nil end
    if TracerConn then pcall(function() TracerConn:Disconnect() end); TracerConn = nil end
    for _, conn in pairs(ESP.CharacterConnections or {}) do
        pcall(function() conn:Disconnect() end)
    end
    ESP.CharacterConnections = {}
    for plr in pairs(ESP.Objects) do ESPDestroy(plr) end
end

    b.Size = UDim2.new(0,55,0,46)
    b.BackgroundColor3 = C.BG2
    b.Text = ""
    b.BorderSizePixel = 0
    b.AutoButtonColor = false
    b.ZIndex = 12
    b.ClipsDescendants = true
    Instance.new("UICorner", b).CornerRadius = UDim.new(0,8)
    local bS = Instance.new("UIStroke", b); bS.Thickness = 2; bS.Color = C.Yel; bS.Transparency = 0.4
    local iL = Instance.new("TextLabel", b)
    iL.Size = UDim2.new(1,0,0,22); iL.Position = UDim2.new(0,0,0,4)
    iL.BackgroundTransparency = 1; iL.Text = i; iL.TextSize = 18; iL.TextColor3 = C.Wh
    iL.Font = Enum.Font.GothamBold; iL.ZIndex = 13
    local nL = Instance.new("TextLabel", b)
    nL.Size = UDim2.new(1,0,0,10); nL.Position = UDim2.new(0,0,1,-14)
    nL.BackgroundTransparency = 1; nL.Text = t; nL.TextColor3 = C.Wh
    nL.TextSize = 7; nL.Font = Enum.Font.GothamBold; nL.ZIndex = 13
    local function SA(s)
        if s then
            b.BackgroundColor3 = C.Yel
            iL.TextColor3 = C.BG; nL.TextColor3 = C.BG
            bS.Color = C.Wh; bS.Transparency = 0
        else
            b.BackgroundColor3 = C.BG2
            iL.TextColor3 = C.Wh; nL.TextColor3 = C.Wh
            bS.Color = C.Yel; bS.Transparency = 0.4
        end
    end
    b.MouseButton1Click:Connect(function()
        local ns = cb()
        if ns ~= nil then SA(ns) end
    end)
    SA(false)
    return b
end

local fdg, fs, fp = false, nil, nil
FH.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        fdg = true; fs = i.Position; fp = FM.Position
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if fdg and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - fs
        FM.Position = UDim2.new(fp.X.Scale, fp.X.Offset + d.X, fp.Y.Scale, fp.Y.Offset + d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then fdg = false end
end)
