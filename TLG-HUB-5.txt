local function GOSt(p)
    if not p or not p.Parent then return end
    GOS()
    GO.T=p
    GO.Ang=0
    local mh0=LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if mh0 then GO.SafeY=math.max(mh0.Position.Y+5,50) end
    GO.C=RS.Heartbeat:Connect(function(dt)
        if not GO.T or not GO.T.Parent then GOS()return end
        local tc=GO.T.Character
        local mc=LP.Character
        if not tc or not mc then GOS()return end
        local th=tc:FindFirstChild("HumanoidRootPart")
        local mh=mc:FindFirstChild("HumanoidRootPart")
        local thu=tc:FindFirstChildOfClass("Humanoid")
        local mhu=mc:FindFirstChildOfClass("Humanoid")
        if not th or not mh or not thu or thu.Health<=0 then GOS()return end
        local dist=(th.Position-mh.Position).Magnitude
        if dist>GO.MaxDist then GOS()return end
        if mhu then mhu.PlatformStand=true end
        GO.Ang=GO.Ang+GO.S*dt
        if GO.Ang>math.pi*2 then GO.Ang=GO.Ang-math.pi*2 end
        local targetY=th.Position.Y+GO.Al
        if targetY<GO.SafeY then targetY=GO.SafeY end
        local off=Vector3.new(math.cos(GO.Ang)*GO.D,0,math.sin(GO.Ang)*GO.D)
        local newPos=Vector3.new(th.Position.X+off.X,targetY,th.Position.Z+off.Z)
        mh.CFrame=CFrame.new(newPos,Vector3.new(th.Position.X,targetY,th.Position.Z))
    end)
end
