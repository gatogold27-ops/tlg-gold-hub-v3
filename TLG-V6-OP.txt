local function GS(p)
    local s={Lv=0,HP=0,MHP=0,Gem=0,Cash=0,Pow=0,Melee=0,Weapon=0}
    pcall(function()
        local ch=p.Character
        if ch then local h=ch:FindFirstChildOfClass("Humanoid") if h then s.HP=math.floor(h.Health) s.MHP=math.floor(h.MaxHealth) end end
    end)
    local function sc(f,d)
        if not f or d>4 then return end
        for _,v in ipairs(f:GetChildren()) do
            if v:IsA("IntValue") or v:IsA("NumberValue") then
                local n=v.Name:lower()
                if n=="level" or n=="lvl" then s.Lv=v.Value
                elseif n=="gems" or n=="gem" then s.Gem=v.Value
                elseif n=="cash" or n=="money" then s.Cash=v.Value
                elseif n=="power" or n=="memepowerlevel" then s.Pow=v.Value
                elseif n=="melee" or n=="meleelevel" then s.Melee=v.Value
                elseif n=="sword" or n=="swordlevel" or n=="weapon" then s.Weapon=v.Value end
            elseif v:IsA("Folder") or v:IsA("Configuration") then sc(v,d+1) end
        end
    end
    for _,n in ipairs({"leaderstats","PlayerData","Stats","Data","Values"}) do pcall(function() sc(p:FindFirstChild(n),0) end) end
    return s
end

local TT={On=false,Act=false,Char=nil,CT=nil,Buf={},Held={},Burst={},Cnx={},BC=5,BI=0.008,MAB=0.4,LA=0}
local TTK={[Enum.KeyCode.Z]=true,[Enum.KeyCode.X]=true,[Enum.KeyCode.C]=true,[Enum.KeyCode.V]=true}
local function TTA(k)
    if not TT.Act or TT.Burst[k] then return end
    local t=TT.CT or (TT.Char and TT.Char:FindFirstChildOfClass("Tool"))
    if not t then return end
    TT.Burst[k]=true
    task.spawn(function()
        for i=1,TT.BC do
            if not TT.Act or not t.Parent then break end
            if i>1 then task.wait(TT.BI) end
            pcall(function() t:Activate() end)
        end
        TT.Burst[k]=nil
    end)
end
local function TTE()
    if not TT.On then return end
    TT.On=false TT.Act=false
    for _,c in ipairs(TT.Cnx) do pcall(function() c:Disconnect() end) end
    TT.Cnx={} TT.Buf={} TT.Held={}
    Ntf("⚡ Timing","Off",C.Rd)
end
local function TTS()
    if TT.On then return end
    TT.On=true
    local function SC(c)
        TT.Act=false TT.Char=c TT.CT=nil
        pcall(function() c:WaitForChild("Humanoid",5) end)
        if TT.Char then TT.CT=TT.Char:FindFirstChildOfClass("Tool") end
        TT.Act=true
    end
    table.insert(TT.Cnx,LP.CharacterAdded:Connect(function(c) task.wait(0.1) SC(c) end))
    if LP.Character then SC(LP.Character) end
    table.insert(TT.Cnx,RS.Heartbeat:Connect(function()
        if not TT.Act then return end
        local now=os.clock()
        if TT.Char then local nt=TT.Char:FindFirstChildOfClass("Tool") if nt~=TT.CT then TT.CT=nt end end
        for k,pt in pairs(TT.Buf) do
            if now-pt<=TT.MAB then if not TT.Burst[k] then TT.Buf[k]=nil TTA(k) end else TT.Buf[k]=nil end
        end
    end))
    if not UI.TouchEnabled then
        table.insert(TT.Cnx,UI.InputBegan:Connect(function(inp,gp)
            if gp or not TT.Act then return end
            if TTK[inp.KeyCode] and not TT.Held[inp.KeyCode] then
                TT.Held[inp.KeyCode]=true TT.Buf[inp.KeyCode]=os.clock() TTA(inp.KeyCode)
            end
        end))
        table.insert(TT.Cnx,UI.InputEnded:Connect(function(inp) if TTK[inp.KeyCode] then TT.Held[inp.KeyCode]=nil end end))
    end
    Ntf("⚡ Timing","✅ On",C.Grn)
end

local AIM={On=false,C=nil,Hs=50,Bs=50,R=5000,Atk=true,Rate=0.05,TP=true,TD=5,Face=true,Data={},LA=0,RC=0,RT=os.clock()}
local function SvO(p,pt,t) if not AIM.Data[p] then AIM.Data[p]={} end if not AIM.Data[p][t] then AIM.Data[p][t]={S=pt.Size,T=pt.Transparency,M=pt.Material,C=pt.Color} end end
local function EH(p)
    local ch=p.Character if not ch then return end
    local h=ch:FindFirstChild("Head") if not h then return end
    SvO(p,h,"H")
    pcall(function() h.Size=Vector3.new(AIM.Hs,AIM.Hs,AIM.Hs) h.Transparency=1 h.CanCollide=false h.Material=Enum.Material.Neon h.Color=AC() end)
end
local function EB(p)
    local ch=p.Character if not ch then return end
    local h=ch:FindFirstChild("HumanoidRootPart") if not h then return end
    SvO(p,h,"B")
    pcall(function() h.Size=Vector3.new(AIM.Bs,AIM.Bs,AIM.Bs) h.Transparency=1 h.CanCollide=false end)
end
local function RA()
    for p,d in pairs(AIM.Data) do
        if p.Character then
            for t,v in pairs(d) do
                local pt=p.Character:FindFirstChild(t=="H" and "Head" or "HumanoidRootPart")
                if pt then pcall(function() pt.Size=v.S pt.Transparency=v.T pt.Material=v.M pt.Color=v.C end) end
            end
        end
    end
    AIM.Data={}
end
local function Find()
    local b=nil bd=AIM.R
    local mc=LP.Character if not mc then return nil end
    local mh=mc:FindFirstChild("HumanoidRootPart") if not mh then return nil end
    for _,p in ipairs(P:GetPlayers()) do
        if p~=LP and p.Character then
            local h=p.Character:FindFirstChild("HumanoidRootPart")
            local u=p.Character:FindFirstChildOfClass("Humanoid")
            if h and u and u.Health>0 then
                local d=(h.Position-mh.Position).Magnitude
                if d<bd then bd=d b=p end
            end
        end
    end
    return b
end
local function Atk()
    local mc=LP.Character if not mc then return end
    local t=mc:FindFirstChildOfClass("Tool")
    if t then pcall(function() t:Activate() end) end
end
local function Loop()
    if not AIM.On then return end
    local t=Find() if not t then return end
    EH(t) EB(t)
    if AIM.Face then
        local mc=LP.Character
        if mc then
            local mh=mc:FindFirstChild("HumanoidRootPart")
            local tc=t.Character
            if mh and tc then
                local th=tc:FindFirstChild("HumanoidRootPart")
                if th then pcall(function() mh.CFrame=CFrame.new(mh.Position,Vector3.new(th.Position.X,mh.Position.Y,th.Position.Z)) end) end
            end
        end
    end
    if AIM.TP then
        local mc=LP.Character local tc=t.Character
        if mc and tc then
            local mh=mc:FindFirstChild("HumanoidRootPart")
            local th=tc:FindFirstChild("HumanoidRootPart")
            if mh and th and (th.Position-mh.Position).Magnitude>AIM.TD then
                pcall(function() mh.CFrame=CFrame.new(SmTP(mh.Position,th.Position),th.Position) end)
            end
        end
    end
    if AIM.Atk then
        local n=os.clock()
        if AD.RL then
            if n-AIM.RT>=1 then AIM.RC=0 AIM.RT=n end
            if AIM.RC>=AD.Max then return end
        end
        if n-AIM.LA>=Jit(AIM.Rate) then
            AIM.LA=n AIM.RC=AIM.RC+1
            if math.random()<0.05 then task.wait(RandP()) end
            Atk()
        end
    end
end
local function AimS()
    if AIM.On then return end
    AIM.On=true
    for _,p in ipairs(P:GetPlayers()) do if p~=LP then EH(p) EB(p) end end
    AIM.C=RS.Heartbeat:Connect(Loop)
    Ntf("🎯 Aimbot","✅ On",C.Grn)
end
local function AimE()
    if not AIM.On then return end
    AIM.On=false
    if AIM.C then AIM.C:Disconnect() AIM.C=nil end
    RA() Ntf("🎯 Aimbot","❌ Off",C.Rd)
end
